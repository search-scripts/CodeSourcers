-- Script para botão arrastável com teleporte e órbita
local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- Criar ScreenGui
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "TeleportOrbitGui"
screenGui.ResetOnSpawn = false
screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

-- Criar botão
local button = Instance.new("TextButton")
button.Name = "OrbitButton"
button.Size = UDim2.new(0, 80, 0, 80)
button.Position = UDim2.new(0.85, 0, 0.7, 0)
button.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
button.BorderSizePixel = 0

-- Variáveis para arrastar
local dragging = false
local dragInput
local dragStart
local startPos
local isOrbiting = false

-- Funções de arrastar
local function update(input)
    local delta = input.Position - dragStart
    button.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

button.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = button.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

button.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        update(input)
    end
end)

-- Função para encontrar jogador mais próximo
local function findNearestPlayer()
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then
        return nil
    end
    
    local myPos = character.HumanoidRootPart.Position
    local nearestPlayer = nil
    local shortestDistance = math.huge
    
    for _, otherPlayer in pairs(Players:GetPlayers()) do
        if otherPlayer ~= player then
            local otherCharacter = otherPlayer.Character
            if otherCharacter and otherCharacter:FindFirstChild("HumanoidRootPart") then
                local distance = (otherCharacter.HumanoidRootPart.Position - myPos).Magnitude
                if distance < shortestDistance then
                    shortestDistance = distance
                    nearestPlayer = otherPlayer
                end
            end
        end
    end
    
    return nearestPlayer
end

-- Função de órbita
local function orbitPlayer(targetPlayer)
    if isOrbiting then return end
    isOrbiting = true
    
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then
        isOrbiting = false
        return
    end
    
    local targetCharacter = targetPlayer.Character
    if not targetCharacter or not targetCharacter:FindFirstChild("Humanoid") then
        isOrbiting = false
        return
    end
    
    local targetHumanoid = targetCharacter.Humanoid
    local targetRootPart = targetCharacter:FindFirstChild("HumanoidRootPart")
    
    if not targetRootPart then
        isOrbiting = false
        return
    end
    
    -- Teleportar próximo ao jogador
    local startAngle = 0
    local orbitRadius = 3 -- 1 metro de distância
    local orbitDuration = 2.5 -- 2.5 segundos
    local startTime = tick()
    
    -- Calcular altura do humanoide
    local humanoidHeight = targetHumanoid.HipHeight + 2
    
    -- Desabilitar colisão temporariamente
    for _, part in pairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = false
        end
    end
    
    -- Loop de órbita
    local connection
    connection = RunService.Heartbeat:Connect(function()
        local elapsedTime = tick() - startTime
        
        if elapsedTime >= orbitDuration or not targetRootPart.Parent then
            connection:Disconnect()
            isOrbiting = false
            
            -- Reabilitar colisão
            for _, part in pairs(character:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = true
                end
            end
            
            return
        end
        
        -- Calcular ângulo atual (360 graus em 2.5 segundos)
        local progress = elapsedTime / orbitDuration
        local angle = startAngle + (math.pi * 2 * progress)
        
        -- Calcular nova posição em órbita
        local offsetX = math.cos(angle) * orbitRadius
        local offsetZ = math.sin(angle) * orbitRadius
        
        local targetPos = targetRootPart.Position
        local newPosition = Vector3.new(
            targetPos.X + offsetX,
            targetPos.Y + humanoidHeight,
            targetPos.Z + offsetZ
        )
        
        -- Mover o jogador
        if character:FindFirstChild("HumanoidRootPart") then
            character.HumanoidRootPart.CFrame = CFrame.new(newPosition, targetPos)
        end
    end)
end

-- Evento de clique do botão
button.MouseButton1Click:Connect(function()
    if isOrbiting then
        return
    end
    
    local nearestPlayer = findNearestPlayer()
    
    if nearestPlayer then
        -- Feedback visual
        button.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
        wait(0.1)
        button.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
        
        -- Iniciar órbita
        orbitPlayer(nearestPlayer)
    else
        -- Feedback de erro
        button.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
        wait(0.3)
        button.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    end
end)

-- Função para atualizar visual do botão
local function updateButtonVisual()
    if isPositionA then
        button.Text = "A"
        button.BackgroundColor3 = Color3.fromRGB(138, 43, 226) -- Roxo
        label.Text = "Pos A"
    else
        button.Text = "B"
        button.BackgroundColor3 = Color3.fromRGB(255, 20, 147) -- Rosa
        label.Text = "Pos B"
    end
end

-- Evento de clique do botão
button.MouseButton1Click:Connect(function()
    local targetPosition
    
    -- Determinar qual posição usar
    if isPositionA then
        targetPosition = POSITION_A
    else
        targetPosition = POSITION_B
    end
    
    -- Tentar teleportar
    local success = teleportToPosition(targetPosition)
    
    if success then
        -- Alternar para a próxima posição
        isPositionA = not isPositionA
        
        -- Feedback visual de sucesso
        local originalColor = button.BackgroundColor3
        button.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
        
        -- Efeito de pulso
        local tweenInfo = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        local goal = {Size = UDim2.new(0, 90, 0, 90)}
        local tween = TweenService:Create(button, tweenInfo, goal)
        tween:Play()
        
        tween.Completed:Connect(function()
            local tweenBack = TweenService:Create(button, tweenInfo, {Size = UDim2.new(0, 80, 0, 80)})
            tweenBack:Play()
        end)
        
        wait(0.2)
        
        -- Atualizar visual para a próxima posição
        updateButtonVisual()
    else
        -- Feedback visual de erro
        button.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
        wait(0.3)
        updateButtonVisual()
    end
end)
